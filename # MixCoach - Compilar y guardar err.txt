# MixCoach - Compilar y guardar errores
$source = "C:\Windows\System32\MixCoach"
$dest = "C:\Proyectos\MixCoach"
Write-Host "[1/3] Copiando archivos..." -ForegroundColor Yellow
if (Test-Path $source) { robocopy $source $dest /E /IS /NDL /NFL /NJH /NJS /R:1 /W:1 | Out-Null }
else { Write-Host "No se encuentra $source - asumiendo que ya están en $dest" }
Set-Location $dest
Write-Host "[2/3] Compilando..." -ForegroundColor Yellow
$output = cmake --build build --config Release 2>&1
$errors = ($output | Select-String -Pattern "error [A-Z]+\d+ :" -AllMatches).Matches.Count
$output | Out-File "$dest\errores.txt" -Encoding UTF8
Write-Host "[3/3] $errors error(es) - guardados en errores.txt" -ForegroundColor $(if ($errors -gt 0){"Red"}else{"Green"})
Write-Host "`nAbre errores.txt y pega su contenido en el chat." -ForegroundColor Cyan
pause
